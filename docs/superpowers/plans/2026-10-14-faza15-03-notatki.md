import Foundation
import MeetNotesProviders

/// Szablon podsumowania: rodzaj spotkania → struktura notatki (spec 1.5 §3, decyzja 2).
/// Preset = język + modele; szablon wybiera się przy „Generate notes" i można go zmienić przy regeneracji.
public struct Template: Codable, Equatable, Identifiable, Sendable {
    public var id: String
    public var name: String
    public var instructions: String
    public var builtin: Bool

    public init(id: String = UUID().uuidString, name: String, instructions: String, builtin: Bool = false) {
        self.id = id
        self.name = name
        self.instructions = instructions
        self.builtin = builtin
    }

    /// Kopia do edycji — wbudowane są tylko do odczytu.
    public func duplicate() -> Template { Template(name: "\(name) copy", instructions: instructions) }

    public static let standard = Template(id: "standard", name: "Standard", instructions: """
        1. A one-paragraph summary: what was discussed and what was agreed.
        2. Topics: one "## " heading per topic, with bullet points that keep the concrete details (numbers, names, dates, \
        the options discussed; mark a favored option with "(favored)"). A side issue gets its own heading.
        3. "## Next Steps": one bullet per action — the task in bold, the owner in parentheses (use the name said in the \
        conversation; if nobody was named write "(owner unclear)"), then one sentence of reasoning.
        """, builtin: true)

    public static let oneToOne = Template(id: "one-to-one", name: "1:1", instructions: """
        1. A short summary of the conversation.
        2. "## Updates": what each person reported, with specifics.
        3. "## Feedback and concerns": what was raised, by whom, and how it was received.
        4. "## Next Steps": bullets — the task in bold, the owner in parentheses, one sentence of reasoning.
        """, builtin: true)

    public static let weeklyTeam = Template(id: "weekly-team", name: "Weekly team meeting", instructions: """
        1. A short summary of the meeting.
        2. "## Progress": what moved since last time, per topic, with numbers where given.
        3. "## Blockers and risks": what is stuck, who is affected.
        4. "## Decisions": what was decided and by whom.
        5. "## Next Steps": bullets — the task in bold, the owner in parentheses, one sentence of reasoning.
        """, builtin: true)

    public static let discoveryCall = Template(id: "discovery-call", name: "Customer: Discovery", instructions: """
        1. A short summary: who the customer is and why they called.
        2. "## Needs and pain points": in the customer's own terms, with specifics.
        3. "## Current situation": tools, process, constraints they described.
        4. "## Objections and open questions".
        5. "## Next Steps": bullets — the task in bold, the owner in parentheses, one sentence of reasoning.
        """, builtin: true)

    public static let builtIns: [Template] = [standard, oneToOne, weeklyTeam, discoveryCall]
}

public enum TemplateError: Error, Equatable {
    case builtinIsReadOnly
    case nameTaken(String)
    case missingName
}

/// Własne szablony użytkownika (`templates.json`). Wbudowane żyją w kodzie, nie w pliku.
public struct TemplatesFile: Codable, Equatable, Sendable {
    public var templates: [Template] = []
    public init(templates: [Template] = []) { self.templates = templates }

    public func all() -> [Template] { Template.builtIns + templates }

    public mutating func upsert(_ t: Template) throws {
        guard !t.builtin, !Template.builtIns.contains(where: { $0.id == t.id }) else { throw TemplateError.builtinIsReadOnly }
        let name = t.name.trimmingCharacters(in: .whitespaces)
        guard !name.isEmpty else { throw TemplateError.missingName }
        guard !all().contains(where: { $0.id != t.id && $0.name.caseInsensitiveCompare(name) == .orderedSame }) else {
            throw TemplateError.nameTaken(name)
        }
        var t = t
        t.name = name
        if let i = templates.firstIndex(where: { $0.id == t.id }) { templates[i] = t } else { templates.append(t) }
    }

    public mutating func remove(id: String) { templates.removeAll { $0.id == id } }
}

import Foundation
import MeetNotesProviders
import MeetNotesTranscription

public struct NotePrompt: Equatable, Sendable {
    public var system: String
    public var user: String
    public init(system: String, user: String) { self.system = system; self.user = user }
}

/// Czysta funkcja: szablon + transkrypt + My notes + kontekst → prompt. `nil`, gdy nie ma czego streszczać
/// (pusta albo w całości wykluczona transkrypcja) — wtedy model nie jest wołany (Review Focus #1).
public enum NotePromptBuilder {
    public static func build(template: Template, transcript: Transcript, myNotes: String, context: String,
                             notesLanguage: LanguageCode?) -> NotePrompt? {
        let lines = transcriptLines(transcript)
        guard !lines.isEmpty else { return nil }
        let language = notesLanguage.map { "Write the notes in \($0.englishName)." }
            ?? "Write the notes in the language that predominates in the transcript."
        let system = """
        You write meeting notes from the transcript of a two-channel recording. "Me" is the user's microphone, "Them" is \
        everyone else on the call.

        Rules:
        - Use only what is in the transcript, the user's own notes and the user's context. Never invent facts, names, \
        numbers or decisions.
        - If the transcript is inconsistent, fragmentary or garbled, say so plainly in the notes (for example: "The \
        transcript is too fragmented to extract reliable notes") and list only what is clearly supported. Do not guess.
        - Be concrete: keep numbers, names, dates and the options that were discussed. No generic filler.
        - The user's own notes are priority: weave them into the result.
        - Output Markdown. The FIRST line must be the title, written as "# <short descriptive title>". Nothing may \
        come before it.
        \(language)

        Structure:
        \(template.instructions)
        """
        let user = """
        Context from the user:
        \(orNone(context))

        The user's own notes:
        \(orNone(myNotes))

        Transcript:
        \(lines.joined(separator: "\n"))
        """
        return NotePrompt(system: system, user: user)
    }

    /// `[m:ss] Me|Them: tekst` — bez wypowiedzi pustych i oznaczonych strażnikiem języka.
    public static func transcriptLines(_ t: Transcript) -> [String] {
        t.utterances
            .filter { !$0.excludedFromNotes && !$0.text.trimmingCharacters(in: .whitespacesAndNewlines).isEmpty }
            .sorted { $0.start < $1.start }
            .map { "[\(clock($0.start))] \($0.speaker == .me ? "Me" : "Them"): \($0.text)" }
    }

    static func clock(_ seconds: Double) -> String {
        let i = Int(seconds)
        return i >= 3600 ? String(format: "%d:%02d:%02d", i / 3600, i / 60 % 60, i % 60) : String(format: "%d:%02d", i / 60, i % 60)
    }

    private static func orNone(_ s: String) -> String {
        let t = s.trimmingCharacters(in: .whitespacesAndNewlines)
        return t.isEmpty ? "(none)" : t
    }
}

import Foundation

/// Tytuły spotkania: ręczny i automatyczny (`notes/meta.json`).
public struct NoteMeta: Codable, Equatable, Sendable {
    public var manualTitle: String?
    public var autoTitle: String?
    public init(manualTitle: String? = nil, autoTitle: String? = nil) {
        self.manualTitle = manualTitle
        self.autoTitle = autoTitle
    }
}

public enum NoteTitle {
    public static let untitled = "Untitled meeting"

    /// Tytuł = pierwszy nagłówek `# ` w pierwszych trzech niepustych liniach odpowiedzi (wstęp i ogrodzenie
    /// bloku kodu są tolerowane). Brak nagłówka → `title == nil`, `body` = cała odpowiedź.
    public static func parse(_ response: String) -> (title: String?, body: String) {
        var lines = response.trimmingCharacters(in: .whitespacesAndNewlines).components(separatedBy: "\n")
        if lines.count >= 2, lines.first!.trimmingCharacters(in: .whitespaces).hasPrefix("```"),
           lines.last!.trimmingCharacters(in: .whitespaces) == "```" {
            lines.removeFirst()
            lines.removeLast()
        }
        var seen = 0
        for (i, line) in lines.enumerated() {
            let t = line.trimmingCharacters(in: .whitespaces)
            if t.isEmpty { continue }
            seen += 1
            if t.hasPrefix("# ") {
                let title = clean(String(t.dropFirst(2)))
                let body = lines[(i + 1)...].joined(separator: "\n").trimmingCharacters(in: .whitespacesAndNewlines)
                return (title.isEmpty ? nil : title, body)
            }
            if seen >= 3 { break }
        }
        return (nil, lines.joined(separator: "\n").trimmingCharacters(in: .whitespacesAndNewlines))
    }

    static func clean(_ s: String) -> String {
        s.trimmingCharacters(in: CharacterSet(charactersIn: "*\"“”' ").union(.whitespaces))
    }

    /// Ręczny > auto > tytuł z notatki > „Untitled meeting". Tytuł z notatki nie nadpisuje istniejącego.
    public static func display(meta: NoteMeta, noteTitle: String?) -> String {
        for c in [meta.manualTitle, meta.autoTitle, noteTitle] {
            if let c, !c.trimmingCharacters(in: .whitespaces).isEmpty { return c }
        }
        return untitled
    }
}

/// Nazwa kopii zapasowej `summary-<yyyyMMdd-HHmmss>.json`; przy kolizji (ta sama sekunda) sufiks `-2`, `-3`…
public enum BackupName {
    public static func name(for date: Date, existing: (String) -> Bool) -> String {
        let f = DateFormatter()
        f.locale = Locale(identifier: "en_US_POSIX")
        f.dateFormat = "yyyyMMdd-HHmmss"
        let base = "summary-\(f.string(from: date))"
        var n = 1
        var candidate = "\(base).json"
        while existing(candidate) { n += 1; candidate = "\(base)-\(n).json" }

    }

import Foundation

public struct MarkdownBlock: Equatable {
    public enum Kind: Equatable { case heading(Int), paragraph, bullet, numbered(Int) }
    public var kind: Kind
    public var depth: Int
    public var text: AttributedString
}

/// Markdown → bloki do narysowania w SwiftUI. `AttributedString(markdown:options:)` z `.full` daje `presentationIntent`
/// (nagłówki, listy zagnieżdżone i numerowane); SwiftUI `Text` sam bloków nie rysuje, więc grupujemy `runs` po
/// identyfikatorze bloku. Sprawdzone na Command Line Tools (2026-10-14).
public enum MarkdownBlocks {
    public static func group(_ markdown: String) -> [MarkdownBlock] {
        guard let a = try? AttributedString(markdown: markdown, options: .init(interpretedSyntax: .full)) else {
            return markdown.isEmpty ? [] : [MarkdownBlock(kind: .paragraph, depth: 0, text: AttributedString(markdown))]
        }
        var blocks: [MarkdownBlock] = []
        var currentID: Int?
        for run in a.runs {
            let comps = run.presentationIntent?.components ?? []
            let id = comps.first?.identity
            let piece = AttributedString(a[run.range])
            if let id, id == currentID, !blocks.isEmpty {
                blocks[blocks.count - 1].text.append(piece)
                continue
            }
            currentID = id
            blocks.append(MarkdownBlock(kind: kind(comps), depth: depth(comps), text: piece))
        }
        return blocks
    }

    static func kind(_ comps: [PresentationIntent.IntentType]) -> MarkdownBlock.Kind {
        if case .header(let level)? = comps.first?.kind { return .heading(level) }
        var ordinal = 0
        var found = false
        for c in comps {
            switch c.kind {
            case .listItem(let n) where !found: ordinal = n; found = true
            case .orderedList where found: return .numbered(ordinal)
            case .unorderedList where found: return .bullet
            default: break
            }
        }
        return .paragraph
    }

    static func depth(_ comps: [PresentationIntent.IntentType]) -> Int {
        max(0, comps.filter { if case .listItem = $0.kind { return true } else { return false } }.count - 1)
    }
}

import MeetNotesProviders
import MeetNotesTranscription

/// Wynik korekty obok oryginału: `runs/<runID>/corrected.json`.
public struct CorrectedTranscript: Codable, Equatable, Sendable {
    public var transcript: Transcript
    public var model: String
    public var batches: Int
    public var failedBatches: Int
    public var rejectedLines: Int

    public init(transcript: Transcript, model: String, batches: Int, failedBatches: Int, rejectedLines: Int) {
        self.transcript = transcript
        self.model = model
        self.batches = batches
        self.failedBatches = failedBatches
        self.rejectedLines = rejectedLines
    }
}

public extension RunDirectory {
    var corrected: URL { url.appendingPathComponent("corrected.json") }
}

public enum CorrectionRules {
    /// Poprawiona linia musi mieć 0,6–1,6× długości oryginału — model nie skraca ani nie rozwija wypowiedzi.
    public static let band: ClosedRange<Double> = 0.6...1.6

    public static func accepts(original: String, corrected: String) -> Bool {
        let o = original.trimmingCharacters(in: .whitespaces).count
        let c = corrected.trimmingCharacters(in: .whitespaces).count
        guard o > 0, c > 0 else { return false }
        return band.contains(Double(c) / Double(o))
    }
}

public enum CorrectionBatches {
    /// Linie `[n] tekst`, pakowane po kolei w partie do `maxChars` znaków (jedna wypowiedź nigdy nie jest dzielona).
    public static func split(_ lines: [(n: Int, text: String)], maxChars: Int) -> [[(n: Int, text: String)]] {
        var out: [[(n: Int, text: String)]] = []
        var cur: [(n: Int, text: String)] = []
        var size = 0
        for l in lines {
            if !cur.isEmpty && size + l.text.count > maxChars { out.append(cur); cur = []; size = 0 }
            cur.append(l)
            size += l.text.count
        }
        if !cur.isEmpty { out.append(cur) }
        return out
    }

    public static func render(_ batch: [(n: Int, text: String)]) -> String {
        batch.map { "[\($0.n)] \($0.text)" }.joined(separator: "\n")
    }

    /// Odpowiedź modelu → `numer → tekst`. `nil`, gdy numery nie są DOKŁADNIE oczekiwanymi (liczba, kolejność, brak/nadmiar).
    public static func parse(_ reply: String, expected: [Int]) -> [Int: String]? {
        var got: [(Int, String)] = []
        for raw in reply.components(separatedBy: "\n") {
            let line = raw.trimmingCharacters(in: .whitespaces)
            guard line.hasPrefix("["), let close = line.firstIndex(of: "]"),
                  let n = Int(line[line.index(after: line.startIndex)..<close]) else { continue }
            got.append((n, String(line[line.index(after: close)...]).trimmingCharacters(in: .whitespaces)))
        }
        guard got.map(\.0) == expected else { return nil }
        return Dictionary(uniqueKeysWithValues: got)
    }
}

/// Korekta po transkrypcji (spec 1.5 §4.1): model z presetu poprawia wypowiedzi partiami. Czas, mówca i flagi języka
/// zostają z oryginału (kopiujemy transkrypt i podmieniamy tylko `text`); wypowiedzi z flagą nie idą do modelu.
public struct Corrector: Sendable {
    public let client: OpenAICompatibleClient
    public let assignment: RoleAssignment
    public var maxBatchChars: Int

    public init(client: OpenAICompatibleClient, assignment: RoleAssignment, maxBatchChars: Int = 6_000) {
        self.client = client
        self.assignment = assignment
        self.maxBatchChars = maxBatchChars
    }

    public func system(language: LanguageCode?) -> String {
        var s = """
        You correct speech-to-text transcripts line by line. Fix recognition errors, punctuation and casing. \
        Do not add, remove, merge or reorder content, and do not summarize or translate.
        The input is numbered lines like "[12] text". Reply with exactly the same numbers, one corrected line per \
        input line, in the same order, and nothing else.
        """
        if let language { s += "\nThe transcript language is \(language.englishName)." }
        if let extra = assignment.prompt?.trimmingCharacters(in: .whitespacesAndNewlines), !extra.isEmpty {
            s += "\nExtra instructions from the user: \(extra)"
        }
        return s
    }

    public func correct(_ original: Transcript, language: LanguageCode?) async -> CorrectedTranscript {
        var out = original
        // Numer linii = indeks wypowiedzi + 1; tylko niepuste i niezaflagowane.
        let lines: [(n: Int, text: String)] = original.utterances.enumerated().compactMap { i, u in
            u.excludedFromNotes || u.text.trimmingCharacters(in: .whitespacesAndNewlines).isEmpty ? nil : (i + 1, u.text)
        }
        let batches = CorrectionBatches.split(lines, maxChars: maxBatchChars)
        let system = system(language: language)
        var failed = 0
        var rejected = 0
        for batch in batches {
            if Task.isCancelled { break }
            do {
                let reply = try await client.chat(model: assignment.model, system: system, user: CorrectionBatches.render(batch))
                guard let fixed = CorrectionBatches.parse(reply, expected: batch.map(\.n)) else { failed += 1; continue }
                for l in batch {
                    guard let new = fixed[l.n], CorrectionRules.accepts(original: l.text, corrected: new) else { rejected += 1; continue }
                    out.utterances[l.n - 1].text = new
                }
            } catch {
                failed += 1
            }
        }
        return CorrectedTranscript(transcript: out, model: assignment.model, batches: batches.count,
                                   failedBatches: failed, rejectedLines: rejected)
    }
}

import Foundation

/// Autotytuł po transkrypcji (spec 1.5 §3, decyzja 18): jedno krótkie wywołanie modelu podsumowania, tylko po tytuł.
/// Zapis do `meta.json.autoTitle` wyłącznie gdy nie ma ręcznego ani wcześniejszego automatycznego. Błąd = cicho
/// „Untitled meeting".
public enum AutoTitle {
    public static let maxLength = 80
    /// Trzy próbki po ~2000 znaków: początek, środek i koniec transkryptu (decyzja użytkownika 2026-10-14).
    public static let sampleChars = 2_000

    public static func shouldRun(meta: NoteMeta) -> Bool { meta.manualTitle == nil && meta.autoTitle == nil }

    /// `nil`, gdy nie ma czego nazywać (pusta albo w całości wykluczona transkrypcja).
    public static func prompt(transcript: Transcript, notesLanguage: LanguageCode?) -> NotePrompt? {
        let lines = NotePromptBuilder.transcriptLines(transcript)
        guard !lines.isEmpty else { return nil }
        let text = sample(lines)
        let language = notesLanguage.map { "in \($0.englishName)" } ?? "in the language that predominates in the transcript"
        let system = "You name meetings. Reply with ONE line: a short descriptive title \(language), at most 8 words, no quotes, "
            + "no trailing period. If the transcript is too fragmented to name, reply with exactly: \(NoteTitle.untitled)"
        return NotePrompt(system: system, user: text)
    }

    /// Krótki transkrypt (do 3 × `sampleChars`) idzie w całości; dłuższy — trzy okna: początek, środek, koniec,
    /// rozdzielone linią „[…]". Wiersz należy do okna, jeśli zaczyna się w jego zakresie znaków.
    static func sample(_ lines: [String]) -> String {
        let total = lines.reduce(0) { $0 + $1.count + 1 }
        guard total > sampleChars * 3 else { return lines.joined(separator: "\n") }
        var starts: [Int] = []
        var acc = 0
        for l in lines { starts.append(acc); acc += l.count + 1 }
        let windows: [Range<Int>] = [0..<sampleChars,
                                     (total / 2 - sampleChars / 2)..<(total / 2 + sampleChars / 2),
                                     (total - sampleChars)..<total]
        let parts = windows.map { w in lines.indices.filter { w.contains(starts[$0]) }.map { lines[$0] }.joined(separator: "\n") }
        return parts.filter { !$0.isEmpty }.joined(separator: "\n[…]\n")
    }

    /// Pierwsza niepusta linia odpowiedzi, bez `#`, cudzysłowów, gwiazdek i kropki; do 80 znaków.
    public static func clean(_ reply: String) -> String? {
        guard let first = reply.components(separatedBy: "\n").map({ $0.trimmingCharacters(in: .whitespaces) })
            .first(where: { !$0.isEmpty }) else { return nil }
        var t = first
        while t.hasPrefix("#") { t.removeFirst() }
        t = t.trimmingCharacters(in: CharacterSet(charactersIn: "*\"“”'. ").union(.whitespaces))
        if t.isEmpty || t == NoteTitle.untitled { return nil }
        return String(t.prefix(maxLength))
    }

    /// Zwraca zapisany tytuł albo `nil` (nie trzeba / nie ma czego nazywać / model zawiódł).
    public static func generate(client: OpenAICompatibleClient, model: String, transcript: Transcript,
                                notesLanguage: LanguageCode?, store: NotesStore) async -> String? {
        guard shouldRun(meta: store.readMeta()), let p = prompt(transcript: transcript, notesLanguage: notesLanguage),
              let reply = try? await client.chat(model: model, system: p.system, user: p.user),
              let title = clean(reply) else { return nil }
        var stored = false
        // Ponowne sprawdzenie pod zapisem: w międzyczasie użytkownik mógł wpisać własny tytuł.
        try? store.updateMeta { m in
            if m.manualTitle == nil && m.autoTitle == nil { m.autoTitle = title; stored = true }
        }
        return stored ? title : nil
    }
}

import Foundation
import MeetNotesProviders

/// Który model podsumowania obsłuży to nagranie i skąd (decyzja użytkownika 2026-10-14).
public struct SummaryChoice: Equatable, Sendable {
    public var assignment: RoleAssignment
    public var presetName: String
    public var isFallback: Bool

    /// „Using <model> from preset <nazwa>" tylko dla zastępstwa; model z presetu nagrania nie wymaga wyjaśnień.
    public var label: String? { isFallback ? "Using \(assignment.model) from preset \(presetName)" : nil }
}

public enum SummaryModel {
    /// Kolejność: (1) rola `summary` zamrożonego presetu nagrania, (2) preset domyślny, (3) dowolny inny preset z `summary`.
    /// `nil` tylko wtedy, gdy żaden preset nie ma modelu podsumowania.
    public static func resolve(frozen: Preset?, defaultPresetID: UUID?, presets: [Preset]) -> SummaryChoice? {
        func usable(_ p: Preset) -> RoleAssignment? {
            p.summary.flatMap { $0.model.trimmingCharacters(in: .whitespaces).isEmpty ? nil : $0 }
        }
        if let f = frozen, let a = usable(f) { return SummaryChoice(assignment: a, presetName: f.name, isFallback: false) }
        let candidates = presets.filter { $0.id == defaultPresetID } + presets.filter { $0.id != defaultPresetID }
        for p in candidates {
            if let a = usable(p) { return SummaryChoice(assignment: a, presetName: p.name, isFallback: true) }
        }
        return nil
    }
}

/// Generowanie notatek, korekta, autotytuł i autozapis My notes (spec 1.5 §4.1). Logika w `MeetNotesNotes`;
/// ten model tylko spina ją z interfejsem i trzyma stan „w toku".
@MainActor
@Observable
final class NotesModel {
    enum GenState: Equatable {
        case idle
        case generating
        case failed(String)
    }

    /// Klucz: id sesji (nazwa katalogu).
    var generation: [String: GenState] = [:]
    private(set) var correcting: Set<String> = []
    private(set) var titling: Set<String> = []
    var templates = TemplatesFile()

    /// Podłączane przez `AppModel` (bez cyklu własności).
    var clientFor: (UUID) -> OpenAICompatibleClient? = { _ in nil }
    var presetFor: (URL) -> Preset? = { _ in nil }
    var presetsInfo: () -> (presets: [Preset], defaultID: UUID?) = { ([], nil) }
    var onChange: () -> Void = {}

    private var saveTasks: [String: Task<Void, Never>] = [:]

    func state(_ id: String) -> GenState { generation[id] ?? .idle }

    /// Model podsumowania dla tego nagrania: z jego presetu, a gdy go tam nie ma — z presetu domyślnego (patrz `SummaryModel`).
    func summaryChoice(for item: SessionItem) -> SummaryChoice? {
        let info = presetsInfo()
        return SummaryModel.resolve(frozen: presetFor(item.url), defaultPresetID: info.defaultID, presets: info.presets)
    }

    func isCorrecting(_ id: String) -> Bool { correcting.contains(id) }
    func isTitling(_ id: String) -> Bool { titling.contains(id) }

    // MARK: Notatka na żądanie

    func generate(item: SessionItem, useCorrected: Bool, template: Template, context: String) {
        let id = item.id
        guard state(id) != .generating else { return }
        guard let transcript = item.transcript(corrected: useCorrected) else {
            generation[id] = .failed("This meeting has no transcript yet.")
            return
        }
        guard let choice = summaryChoice(for: item) else {
            generation[id] = .failed("No preset has a summary model. Choose one in Presets.")
            return
        }
        guard let client = clientFor(choice.assignment.providerID) else {
            generation[id] = .failed("Couldn't reach the summary provider. Check Settings.")
            return
        }
        generation[id] = .generating
        let generator = NoteGenerator(client: client, model: choice.assignment.model)
        // Język notatki z presetu NAGRANIA (jego język i „Notes language"), także gdy model pochodzi z innego presetu.
        let language = presetFor(item.url)?.effectiveNotesLanguage
        let session = item.url
        Task {
            do {
                _ = try await generator.generate(session: session, template: template, transcript: transcript,
                                                 context: context, notesLanguage: language)
                self.generation[id] = .idle
            } catch NoteGenerationError.nothingToSummarize {
                self.generation[id] = .failed("Nothing to summarize: the transcript has no usable text.")
            } catch NoteGenerationError.failed(let message) {
                self.generation[id] = .failed(message)
            } catch {
                self.generation[id] = .failed("\(error)")
            }
            self.onChange()
        }
    }

    // MARK: Po transkrypcji: korekta (gdy preset ma model) i autotytuł (gdy ma model podsumowania)

    func afterTranscription(sessionURL: URL, runID: String, snapshot: RunSnapshot, transcript: Transcript) async {
        let id = sessionURL.lastPathComponent
        var working = transcript
        if let corr = snapshot.preset.correction, !corr.model.isEmpty, let client = clientFor(corr.providerID) {
            correcting.insert(id)
            onChange()
            let result = await Corrector(client: client, assignment: corr).correct(transcript, language: snapshot.language)
            correcting.remove(id)
            if result.batches > 0 {
                try? JSONFile.save(result, to: RunDirectory(session: sessionURL, runID: runID).corrected)
                working = result.transcript
            }
            onChange()
        }
        if let sum = snapshot.preset.summary, !sum.model.isEmpty, let client = clientFor(sum.providerID) {
            titling.insert(id)
            onChange()
            _ = await AutoTitle.generate(client: client, model: sum.model, transcript: working,
                                         notesLanguage: snapshot.preset.effectiveNotesLanguage, store: NotesStore(session: sessionURL))
            titling.remove(id)
            onChange()
        }
    }

    // MARK: My notes i tytuł

    /// Autozapis z opóźnieniem: ostatnia zmiana wygrywa, zapis 0,6 s po ostatnim znaku.
    func saveMyNotes(_ text: String, for item: SessionItem) {
        let id = item.id
        let session = item.url
        saveTasks[id]?.cancel()
        saveTasks[id] = Task {
            try? await Task.sleep(for: .milliseconds(600))
            guard !Task.isCancelled else { return }
            try? NotesStore(session: session).writeMyNotes(text)
        }
    }

    func setManualTitle(_ title: String, for item: SessionItem) {
        let t = title.trimmingCharacters(in: .whitespacesAndNewlines)
        try? NotesStore(session: item.url).updateMeta { $0.manualTitle = t.isEmpty ? nil : t }
        onChange()
    }
}

/// Czy katalog sesji to ten, który właśnie nagrywamy. Porównanie po nazwie katalogu: URL z `contentsOfDirectory`

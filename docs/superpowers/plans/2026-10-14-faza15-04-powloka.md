```swift
import Foundation
import MeetNotesTranscription

/// Jeden status sesji dla listy (spec 1.5 §6). Kolejność warunków = kolejność cyklu życia:
/// nagrywanie → transkrypcja → korekta → tytuł → … Trwające nagranie NIGDY nie jest „Recovered".
public enum SessionStatus: Equatable, Sendable {
    case recording
    case transcribing(Double)
    case correcting
    case titling
    case recovered
    case partial(gaps: Int)
    case interrupted
    case notesReady
    case flagged(Int)
    case transcriptReady
    case notTranscribed

    /// Czy katalog sesji to ten, który właśnie nagrywamy. Porównanie po nazwie katalogu: URL z `contentsOfDirectory`
    /// bywa zapisany z ukośnikiem na końcu, a URL zbudowany przez `appendingPathComponent` — bez.
    public static func isRecording(_ session: URL, recordingDirectory: URL?) -> Bool {
        guard let recordingDirectory else { return false }
        return session.lastPathComponent == recordingDirectory.lastPathComponent
    }

    public static func of(isRecording: Bool, progress: Double?, correcting: Bool, titling: Bool, crashed: Bool,
                          transcript: Transcript?, hasRun: Bool, hasNotes: Bool) -> SessionStatus {
        if isRecording { return .recording }
        if let p = progress, p < 1 { return .transcribing(p) }
        if correcting { return .correcting }
        if titling { return .titling }
        if crashed { return .recovered }
        if let t = transcript {
            if !t.failedChunks.isEmpty { return .partial(gaps: t.failedChunks.count) }
            if hasNotes { return .notesReady }
            if t.languageFlagCount > 0 { return .flagged(t.languageFlagCount) }
            return .transcriptReady
        }
        return hasRun ? .interrupted : .notTranscribed
    }
}
```

```python
import json, re, sys, pathlib

def letters(s):
    return [c for c in s if c.isalpha()]

def foreign_share(s):
    ls = letters(s)
    foreign = sum(1 for c in ls if not (ord(c) <= 0x24F or 0x1E00 <= ord(c) <= 0x1EFF))
    return foreign / max(1, len(ls)), len(ls)

def guard_reference(s):  # ta sama reguła co LanguageGuard.nonLatinScript
    share, n = foreign_share(s)
    return n >= 4 and share > 0.5

def load(path):
    t = json.loads(pathlib.Path(path).read_text())
    return [x for x in t["utterances"] if x["text"].strip()]

def stats(path):
    u = load(path)
    return {
        "utterances": len(u),
        "words": sum(len(x["text"].split()) for x in u),
        "guard_reference": sum(1 for x in u if guard_reference(x["text"])),
        "flagged_by_app": sum(1 for x in u if x.get("languageMismatch")),
        "cyrillic_any": sum(1 for x in u if re.search(r"[\u0400-\u04FF]", x["text"])),
        "polish_letters": sum(1 for x in u if re.search(r"[ąćęłńóśźż]", x["text"].lower())),
    }

def dump_sample(path, n_near=15, n_even=15):
    """Do ręcznego oznaczenia: WSZYSTKIE oflagowane + jawnie dobrana próba nieoflagowanych
    (n_near o najwyższym udziale obcych liter — „prawie" oflagowane — i n_even równomiernie rozłożonych)."""
    u = load(path)
    flagged = [(i, x) for i, x in enumerate(u) if x.get("languageMismatch")]
    rest = [(i, x) for i, x in enumerate(u) if not x.get("languageMismatch")]
    near = sorted(rest, key=lambda p: -foreign_share(p[1]["text"])[0])[:n_near]
    step = max(1, len(rest) // n_even)
    even = rest[::step][:n_even]
    for tag, rows in (("FLAGGED", flagged), ("NEAR", near), ("EVEN", even)):
        for i, x in rows:
            print(f"{tag}\t{i}\t{foreign_share(x['text'])[0]:.2f}\t{x['text']}")

if __name__ == "__main__":
    mode, *runs = sys.argv[1:]
    for run in runs:
        print(run, stats(f"{run}/transcript.json") if mode == "stats" else "")
        if mode == "sample":
            dump_sample(f"{run}/transcript.json")
```

```swift
import Foundation

/// Kod `OSStatus` → zdanie dla człowieka. Czteroznakowe kody (`'wht?'`) rozpoznajemy po bajtach; resztę podajemy liczbą.
public enum OSStatusText {
    public static func fourCC(_ code: Int) -> String? {
        let u = UInt32(truncatingIfNeeded: code)
        let bytes = [24, 16, 8, 0].map { UInt8((u >> UInt32($0)) & 0xFF) }
        guard bytes.allSatisfy({ $0 >= 0x20 && $0 < 0x7F }) else { return nil }
        return String(decoding: bytes, as: UTF8.self)
    }

    public static func describe(_ code: Int) -> String {
        switch fourCC(code) {
        case "wht?": "unspecified audio file error — the file is missing or unreadable"
        case "typ?": "unsupported audio file type"
        case "fmt?": "unsupported audio data format"
        case "dta?": "invalid audio data"
        case let c?: "audio error '\(c)' (\(code))"
        case nil: "audio error \(code)"
        }
    }
}
```
